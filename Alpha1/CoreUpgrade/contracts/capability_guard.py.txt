"""
Capability Guard

Enforces strict runtime permission checks against contract capability sets 
to prevent unauthorized cross-domain actions.
"""

from typing import Any
from contracts.capability import ContractCapability
from contracts.exceptions import ContractValidationError


class CapabilityGuard:

    @staticmethod
    def require(contract: Any, capability: ContractCapability) -> None:
        """
        Asserts that a contract possesses a specific runtime capability.
        Raises ContractValidationError if the required capability is absent.
        """
        cap_enum = ContractCapability(capability)
        if not hasattr(contract, "capabilities") or cap_enum not in contract.capabilities:
            contract_id = getattr(contract, "contract_id", "unknown")
            raise ContractValidationError(
                f"Capability violation: contract {contract_id} lacks required capability '{cap_enum.value}'."
            )
